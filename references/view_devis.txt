function lancerDecoupage(baseUrl) {
    const token   = Math.random().toString(36).slice(2, 11);
    const pollUrl = 'https://atelierpro.vortech-x.com/alum/pdf-ready/:token'.replace(':token', token);

    const loader  = document.getElementById('pdfLoader');
    const timerEl = document.getElementById('pdfTimer');
    const barEl   = document.getElementById('pdfProgressBar');

    // Réinitialiser la barre
    barEl.style.background = 'linear-gradient(90deg,#f59e0b,#fbbf24)';
    barEl.style.width = '0%';
    loader.style.display = 'flex';

    // Déclencher le téléchargement sans quitter la page
    const a = document.createElement('a');
    a.href = baseUrl + '?token=' + token;
    a.style.display = 'none';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);

    // Chronomètre + barre de progression estimée
    let elapsed = 0;
    const tick = setInterval(() => {
        elapsed++;
        timerEl.textContent = elapsed;
        const pct = 90 * (1 - Math.exp(-elapsed / 25));
        barEl.style.width = pct + '%';
    }, 1000);

    // Polling AJAX toutes les 800ms
    const poll = setInterval(() => {
        fetch(pollUrl, { headers: { 'X-Requested-With': 'XMLHttpRequest' } })
            .then(r => r.json())
            .then(data => {
                if (data.ready) {
                    clearInterval(poll);
                    clearInterval(tick);
                    barEl.style.width = '100%';
                    barEl.style.background = 'linear-gradient(90deg,#22c55e,#4ade80)';
                    setTimeout(() => { loader.style.display = 'none'; }, 700);
                }
            })
            .catch(() => {}); // silencieux si réseau
    }, 800);

    // Sécurité : fermer après 3 minutes quoi qu'il arrive
    setTimeout(() => {
        clearInterval(poll);
        clearInterval(tick);
        loader.style.display = 'none';
    }, 180000);
}
