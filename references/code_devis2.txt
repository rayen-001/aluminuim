        protoToggle(item, key, ref, maxBars) {
            const sel = item[key];
            if (sel[ref] !== undefined) {
                delete sel[ref];
                const left = Object.keys(sel);
                if (left.length === 1) {
                    sel[left[0]] = maxBars;
                }
            } else {
                if (Object.keys(sel).length >= maxBars) { return; }
                const remaining = maxBars - this.protoTotal(item, key);
                if (remaining >= 1) {
                    sel[ref] = remaining;
                } else {
                    const biggest = Object.keys(sel).sort((a, b) => sel[b] - sel[a])[0];
                    if (biggest && sel[biggest] > 1) { sel[biggest]--; }
                    sel[ref] = 1;
                }
            }
        },

        protoAdjust(item, key, ref, delta, maxBars) {
            const sel = item[key];
            const totalOthers = this.protoTotal(item, key) - Number(sel[ref]);
            sel[ref] = Math.min(Math.max(Number(sel[ref]) + delta, 1), maxBars - totalOthers);
        },

        // origine, et « + » les concaténerait au lieu de les additionner.

            return Object.values(item[key]).reduce((sum, qty) => sum + Number(qty), 0);

            // Réinitialiser store/mousti à chaque changement de type produit
            this.items[index].store_enabled  = false;
            this.items[index].mousti_enabled = false;

            // Composition (dormant/traverse/parclose) : fournie par l'API types
            const comp = typeInfo ? (typeInfo.composition || null) : null;
            this.items[index].composition = comp;
            this.items[index].has_cremone = typeInfo ? !!typeInfo.has_cremone : false;
            this.items[index].cremone_id = '';
            if (comp) {
                this.items[index].comp_ouvrant_ref  = comp.ouvrant.default;
                this.items[index].comp_dormant_ref  = comp.dormant.default;
                this.items[index].comp_traverse_ref = comp.traverse.default;
                this.items[index].comp_parclose_ref = this.parcloseDefault(this.items[index]);
                this.items[index].comp_meneau_ref   = comp.meneau.default;
                this.items[index].comp_couvre_joint_ref = comp.couvre_joint.default;
                if (comp.coulissant) {
                    this.items[index].comp_lateral_qty = { [comp.lateral.default]: comp.lateral.count };
                    this.items[index].comp_central_qty = { [comp.central.default]: comp.central.count };
                } else {
                    this.items[index].comp_lateral_qty = {};
                    this.items[index].comp_central_qty = {};
                }
                this.items[index].comp_traverse_enabled = false;
                this.items[index].comp_seuil_ref = (this.seuilOptions(this.items[index])[0] || '');
            } else {
                this.items[index].comp_ouvrant_ref  = '';
                this.items[index].comp_dormant_ref  = '';
                this.items[index].comp_traverse_ref = '';
                this.items[index].comp_parclose_ref = '';
                this.items[index].comp_meneau_ref   = '';
                this.items[index].comp_couvre_joint_ref = '';
                this.items[index].comp_lateral_qty  = {};
                this.items[index].comp_central_qty  = {};
            }

            // PROTOTYPE (front only) : Composition coulissant — famille "Coulissante TPR S67",
            // types "... N vantaux". 2 barres latérales + (N-1)*2 barres centrales.
            // Multi-choix par rôle plafonné au nb de barres. Aucun envoi backend.
            const protoFamName = this.familyNames[this.items[index].family_id] || '';
            const protoVantMatch = typeInfo ? typeInfo.name.match(/(\d+)\s*vantaux/i) : null;
            if (false && protoFamName === 'Coulissante TPR S67' && protoVantMatch) {
                const nVantaux = parseInt(protoVantMatch[1]);
                const nbCentral = (nVantaux - 1) * 2;
                this.items[index].proto_coulissant = { nVantaux: nVantaux, nbCentral: nbCentral };
                this.items[index].proto_dormant_ref = this.protoCoulissantRefs.dormant[0];
                this.items[index].proto_parclose_ref = this.protoCoulissantRefs.parclose[0];
                this.items[index].proto_cj_ref = this.protoCoulissantRefs.couvre_joint[0];
                this.items[index].comp_lateral_qty = { '67104': 2 };
                this.items[index].comp_central_qty = { '67105': nbCentral };
            } else {
                this.items[index].proto_coulissant = null;
                this.items[index].proto_dormant_ref = '';
                this.items[index].proto_parclose_ref = '';
                this.items[index].proto_cj_ref = '';
                this.items[index].comp_lateral_qty = {};
                this.items[index].comp_central_qty = {};
            }

            // Standalone Store : activer automatiquement le store
            if (isStandaloneStore) {
                this.items[index].store_enabled = true;
                return;
            }

            // Standalone Moustiquaire : activer automatiquement la moustiquaire
            if (isStandaloneMousti) {
                this.items[index].mousti_enabled = true;
                this.items[index].mousti_hauteur = this.items[index].hauteur;
                this.items[index].mousti_largeur = this.items[index].largeur;
                return;
            }

            // Reset couvre joint si valeur incompatible avec une porte
            if (isPorte && ['Bas', 'Haut et Bas'].includes(this.items[index].couvre_joint_type)) {
                this.items[index].couvre_joint_type = 'Droite';
            }

            if (isGardeCorps) {
                const name = typeInfo.name;
                this.items[index].gc_has_poteaux = ['Garde corps Rond Linaire', 'Garde corps Rond Vitré', 'Garde corps Corpsen', 'Garde corps Corpsen Sabot'].includes(name);
                // Garde corps vitrés : remplissage + motif sélectionnables
                this.items[index].gc_has_vitrage = ['Garde corps Rond Vitré', 'Garde corps Corpsen', 'Garde corps Corpsen Sabot'].includes(name);
                this.items[index].requires_remplissage = this.items[index].gc_has_vitrage;
                this.items[index].gc_has_lignes  = name === 'Garde corps Rond Linaire';
                this.items[index].gc_show_ongle    = name !== 'Pass Main';
                this.items[index].gc_ongle_checkbox = ['Garde corps Rond Vitré', 'Garde corps Corpsen', 'Garde corps Corpsen Sabot'].includes(name);
                // Par défaut : checkbox cochée (90°)
                if (this.items[index].gc_ongle_checkbox) {
                    this.items[index].gc_ongle = '90°';
                }
                this.items[index].gc_has_coudes  = true;
                // Defaults pour types sans poteaux
                if (!this.items[index].gc_has_poteaux) {
                    this.items[index].gc_nb_poteaux = 0;
                    this.items[index].gc_nb_lignes  = 0;
                }
                return;
            } else if (isChassi) {
                // Charge les refs cadre/socle/montant/traverse
                const response = await fetch(`https://atelierpro.vortech-x.com/alum/api/chassi-fix-refs?type_id=${typeId}`);
                const refs = await response.json();
                this.items[index].chassi_refs = refs;
                this.items[index].chassi_cadre_ref    = refs.cadre[0]    ?? '';
                this.items[index].chassi_socle_ref    = refs.socle[0]    ?? '';
                this.items[index].chassi_montant_ref  = refs.montant[0]  ?? '';
                this.items[index].chassi_traverse_ref = refs.traverse[0] ?? '';
                this.items[index].chassi_montant_enabled  = false;
                this.items[index].chassi_traverse_enabled = false;
            } else {
                // Produit standard : charge les options normales
                const response = await fetch(`https://atelierpro.vortech-x.com/alum/api/options?type_id=${typeId}`);
                const options = await response.json();
                this.items[index].options = options;

                // Partie fixe indépendante (ex: Coulissant - Partie fixe droite/gauche/haut/bas)
                this.items[index].is_partie_fix   = !!options.is_partie_fix;
                this.items[index].partie_fix_label = options.partie_fix_label || '';
                if (options.is_partie_fix && options.partie_fix_def_dim) {
                    this.items[index].pf_dim_1 = options.partie_fix_def_dim;
                }

                if (options.vitrage && options.vitrage.length > 0) {
                    this.items[index].vitrage_type = 'simple';
                }
                if (options.ouverture && options.ouverture.length > 0) {
                    this.items[index].ouverture_type = options.ouverture[0];
                }
                // Serrure : sélectionner le premier choix par défaut
                if (options.serrure && options.serrure.length > 0) {
                    const serrureTags = ['Serrure montant', 'Crémone', 'Serrure traverse'];
                    this.items[index].supplements = [
                        ...this.items[index].supplements.filter(t => !serrureTags.map(s => s.toLowerCase()).includes(t.toLowerCase())),
                        options.serrure[0]
                    ];
                }
            }
        },

        onVitrageChange(index) {
            if (this.items[index].vitrage_type === 'double' && this.doubleVitrageMotifId) {
                this.items[index].motif_id = this.doubleVitrageMotifId;
            } else if (this.items[index].vitrage_type === 'simple' && this.items[index].motif_id == this.doubleVitrageMotifId) {
                this.items[index].motif_id = '';
            }
            // La parclose dépend du vitrage : revenir à celle par défaut
            if (this.items[index].composition) {
                this.items[index].comp_parclose_ref = this.parcloseDefault(this.items[index]);
            }
        },

        // Certains ouvrants (ex. 40404) intègrent la parclose : pas de choix à faire
        showParclose(item) {
            if (!item.composition || item.composition.coulissant) { return false; }
            const elim = item.composition.ouvrant.eliminate_parclose || [];
            return !elim.includes(item.comp_ouvrant_ref);
        },

        parcloseOptions(item) {
            if (!item.composition) { return []; }
            const key = item.vitrage_type === 'double' ? 'double' : 'simple';
            return item.composition.parclose[key].options;
        },

        parcloseDefault(item) {
            if (!item.composition) { return ''; }
            const key = item.vitrage_type === 'double' ? 'double' : 'simple';
            return item.composition.parclose[key].default;
        },

        hasSerrureTraverseSel(item) {
            return (item.supplements || []).some(s => (s || '').toLowerCase().includes('serrure travers'));
        },

        // Coupe du profilé : une image par référence dans public/images/profils.
        // Gamme TPR uniquement — les réfs Alu Eco (AE_) n'ont pas de vignette.
        // Une réf sans image reçoit une coupe générique, à remplacer au fil de l'eau.
        profilImage(ref) {
            // Les réfs du châssis fixe arrivent en nombres (clés PHP numériques),
            // celles de la composition en chaînes : normaliser avant tout traitement.
            const code = String(ref ?? '').trim();
            if (code === '' || code.startsWith('AE_')) { return null; }
            const base = 'https://atelierpro.vortech-x.com/profils/';
            const fichier = this.profilsAvecImage[code.toLowerCase().replace(/\s+/g, '')];
            return base + (fichier || '_defaut.svg');
        },

        // Fast Lock : le choix du nombre de points ne vaut que si l'option est cochée
        hasFastLock(item) {
            return (item.supplements || []).some(s => (s || '').toLowerCase() === 'fast lock');
        },

        hasCremoneSel(item) {
            return (item.supplements || []).some(s => (s || '').toLowerCase() === 'crémone');
        },

        hasPartieFixe(item) {
            return item.partie_fixe_type && item.partie_fixe_type !== 'Sans';
        },

        // Coulissant porte : options de seuil bas si le dormant choisi est composite
        seuilOptions(item) {
            if (!item.composition || !item.composition.coulissant || !item.composition.dormant_composite) { return []; }
            return item.composition.dormant_composite[item.comp_dormant_ref] || [];
        },

        // Couvre joint obligatoire si le dormant choisi est un déclencheur (dormant nu),
        // SAUF si "Sans couvre joint" + "Tous" est coché (pas de couvre joint du tout).
        needsCouvreJoint(item) {
            if (item.sans_couvre_joint && item.couvre_joint_type === 'Tous') {
                return false;
            }
            return !!(item.composition
                && item.composition.couvre_joint
                && item.composition.couvre_joint.triggers.includes(item.comp_dormant_ref));
        },

        toggleSupplement(index, value, event) {
            const sups = this.items[index].supplements;
            if (event.target.type === 'radio') {
                // Radio: remove all other serrure tags first, then add selected
                const serrureTags = ['Serrure montant', 'Crémone', 'Serrure traverse'];
                this.items[index].supplements = sups.filter(s => !serrureTags.map(t => t.toLowerCase()).includes(s.toLowerCase()));
                this.items[index].supplements.push(value);
            } else if (event.target.checked) {
                if (!sups.includes(value)) sups.push(value);
            } else {
                const i = sups.indexOf(value);
                if (i > -1) sups.splice(i, 1);
            }
        },
    };
}
